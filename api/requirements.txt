#
# Deliberately NOT the project's root requirements.txt: that file carries the
# research stack (mlflow, fairlearn, jupyter, streamlit, pytest, linters) which
# the API never imports and which would roughly double the image.

# Versions are pinned to the set verified working together in development, so a
# rebuild months from now produces the same container rather than whatever pip
# resolves that day.

# --- numerical core ---------------------------------------------------------

# --- model loading and inference --------------------------------------------

joblib==1.5.2
scikit-learn==1.7.2
xgboost==3.4.1

# --- explanations -----------------------------------------------------------
